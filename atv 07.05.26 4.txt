package com.exemplo.games.entity;

import jakarta.persistence.*;

@Entity
public class Jogo {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    private String nome;
    private String genero;
    private Double preco;

    public Jogo() {
    }

    public Jogo(Long id, String nome, String genero, Double preco) {
        this.id = id;
        this.nome = nome;
        this.genero = genero;
        this.preco = preco;
    }

    public Long getId() {
        return id;
    }

    public String getNome() {
        return nome;
    }

    public void setNome(String nome) {
        this.nome = nome;
    }

    public String getGenero() {
        return genero;
    }

    public void setGenero(String genero) {
        this.genero = genero;
    }

    public Double getPreco() {
        return preco;
    }

    public void setPreco(Double preco) {
        this.preco = preco;
    }

    public void setId(Long id) {
        this.id = id;
    }
}

3. Classe Mapper — JogoMapper

Responsável pela conversão entre Entity e DTO.

package com.exemplo.games.mapper;

import com.exemplo.games.dto.JogoRequestDTO;
import com.exemplo.games.dto.JogoResponseDTO;
import com.exemplo.games.entity.Jogo;

public class JogoMapper {

    public static Jogo toEntity(JogoRequestDTO dto) {

        Jogo jogo = new Jogo();

        jogo.setNome(dto.getNome());
        jogo.setGenero(dto.getGenero());
        jogo.setPreco(dto.getPreco());

        return jogo;
    }

    public static JogoResponseDTO toDTO(Jogo jogo) {

        return new JogoResponseDTO(
                jogo.getId(),
                jogo.getNome(),
                jogo.getGenero(),
                jogo.getPreco()
        );
    }
}

5. Service

Agora a conversão acontece na camada de service usando o mapper.

package com.exemplo.games.service;

import com.exemplo.games.dto.JogoRequestDTO;
import com.exemplo.games.dto.JogoResponseDTO;
import com.exemplo.games.entity.Jogo;
import com.exemplo.games.mapper.JogoMapper;
import com.exemplo.games.repository.JogoRepository;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.List;

@Service
public class JogoService {

    @Autowired
    private JogoRepository repository;

    public JogoResponseDTO cadastrar(JogoRequestDTO dto) {

        Jogo jogo = JogoMapper.toEntity(dto);

        Jogo salvo = repository.save(jogo);

        return JogoMapper.toDTO(salvo);
    }

    public List<JogoResponseDTO> listar() {

        return repository.findAll()
                .stream()
                .map(JogoMapper::toDTO)
                .toList();
    }

    public JogoResponseDTO buscarPorId(Long id) {

        Jogo jogo = repository.findById(id)
                .orElseThrow(() -> new RuntimeException("Jogo não encontrado"));

        return JogoMapper.toDTO(jogo);
    }
}
6. Controller

O controller trabalha apenas com DTOs.

package com.exemplo.games.controller;

import com.exemplo.games.dto.JogoRequestDTO;
import com.exemplo.games.dto.JogoResponseDTO;
import com.exemplo.games.service.JogoService;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/jogos")
public class JogoController {

    @Autowired
    private JogoService service;

    @PostMapping
    public JogoResponseDTO cadastrar(@RequestBody JogoRequestDTO dto) {
        return service.cadastrar(dto);
    }

    @GetMapping
    public List<JogoResponseDTO> listar() {
        return service.listar();
    }

    @GetMapping("/{id}")
    public JogoResponseDTO buscarPorId(@PathVariable Long id) {
        return service.buscarPorId(id);
    }
}

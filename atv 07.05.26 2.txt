Entidade
Jogo.java
package com.exemplo.jogos.entity;import jakarta.persistence.*;@Entitypublic class Jogo {    @Id    @GeneratedValue(strategy = GenerationType.IDENTITY)    private Long id;    private String titulo;    private String estudio;    private Integer anoLancamento;    private Double preco;    public Jogo() {    }    public Jogo(Long id, String titulo, String estudio,                Integer anoLancamento, Double preco) {        this.id = id;        this.titulo = titulo;        this.estudio = estudio;        this.anoLancamento = anoLancamento;        this.preco = preco;    }    public Long getId() {        return id;    }    public void setId(Long id) {        this.id = id;    }    public String getTitulo() {        return titulo;    }    public void setTitulo(String titulo) {        this.titulo = titulo;    }    public String getEstudio() {        return estudio;    }    public void setEstudio(String estudio) {        this.estudio = estudio;    }    public Integer getAnoLancamento() {        return anoLancamento;    }    public void setAnoLancamento(Integer anoLancamento) {        this.anoLancamento = anoLancamento;    }    public Double getPreco() {        return preco;    }    public void setPreco(Double preco) {        this.preco = preco;    }}

Controller
JogoController.java
package com.exemplo.jogos.controller;import com.exemplo.jogos.entity.Jogo;import com.exemplo.jogos.repository.JogoRepository;import org.springframework.beans.factory.annotation.Autowired;import org.springframework.web.bind.annotation.*;import java.util.List;import java.util.Optional;@RestController@RequestMapping("/jogos")public class JogoController {    @Autowired    private JogoRepository repository;    // POST /jogos    @PostMapping    public Jogo cadastrar(@RequestBody Jogo jogo) {        return repository.save(jogo);    }    // GET /jogos    @GetMapping    public List<Jogo> listarTodos() {        return repository.findAll();    }    // GET /jogos/{id}    @GetMapping("/{id}")    public Optional<Jogo> buscarPorId(@PathVariable Long id) {        return repository.findById(id);    }}
